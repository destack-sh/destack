// expressions / patterns / array / array patterns bind elements
let [first, second] = [1, 2];
// <- keyword.declaration.type.ds

// expressions / patterns / array / array destructuring requires an initializer
let [value]: number[];

// expressions / patterns / array / array patterns need iterable values
let [value] = 1;

// expressions / patterns / array / array defaults fill absent elements
let [value = 1] = [];
value satisfies int32;

// expressions / patterns / array / array defaults check element types
let [value = "no"]: int32[] = [];

// expressions / patterns / array / array rest binds tails
let [head, ...tail] = [1, 2, 3];

tail satisfies number[];

// expressions / patterns / array / slice rest keeps slices

tail satisfies [int32];
