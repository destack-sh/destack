// expressions / dispatch / operators / binary operators use the left receiver
struct Left {}
// <- keyword.declaration.type.ds
struct Right {}

extension of Right implements Add<Left> {
    type Output = Right;

    add(other: Left): this.Output {
        return Right {};
    }
}

declare let left: Left;
declare let right: Right;

left + right;

// expressions / dispatch / operators / reversed operands need reversed implementations
struct Vector2 {
    x: int32;
    y: int32;
}
struct Scalar {
    value: int32;
}

extension of Vector2 implements Add<Scalar> {
    type Output = Vector2;

    add(other: Scalar): this.Output {
        return this;
    }
}

extension of Scalar implements Add<Vector2> {
    type Output = Vector2;

    add(other: Vector2): this.Output {
        return other;
    }
}

declare let vector: Vector2;
declare let scalar: Scalar;

const padded = vector + scalar;
const scaled = scalar + vector;

padded satisfies Vector2;
scaled satisfies Vector2;
