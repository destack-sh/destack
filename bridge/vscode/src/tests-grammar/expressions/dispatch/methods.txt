// expressions / dispatch / methods / methods use declaration order
class Parser {
    parse(value: number): "broad" {
        return "broad";
    }

    parse(value: 1 | 2): "narrow" {
        return "narrow";
    }
}

const parser = new Parser();
const selected = parser.parse(1);
selected satisfies "broad";
