// SYNTAX TEST "source.ds"

// expressions / using / async / await using accepts async resources
class Connection implements AsyncDispose {
    asyncDispose(): Promise<void> {
        Promise.resolve()
    }
}

async function run(): Promise<void> {
    await using connection = new Connection();
    connection satisfies Connection;
}

// expressions / using / async / await using accepts sync resources
class File implements Dispose {
    dispose(): void {}
}

async function run(): Promise<void> {
    await using file = new File();
    file satisfies File;
}

// expressions / using / async / await using accepts nullish resources
async function run(): Promise<void> {
    await using missing = null;
    await using absent = undefined;
}

// expressions / using / async / await using rejects non disposable values
async function run(): Promise<void> {
    await using value = 1;
}

function run(): void {
    await using value = null;
}
