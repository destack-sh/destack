// expressions / comptime / try / comptime try can catch propagated failures
function parse(value: int): Result<int, string> {
    if (value > 0) {
        Result.ok(value)
    } else {
        Result.err("bad")
    }
}

        parse(0)?
    } catch (error) {
        error satisfies string;
        1
    }
};
