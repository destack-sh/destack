// expressions / comptime / scope / comptime locals can be mutated
const width = comptime {
    let value = 4;
    value *= 2;
    value
};

width satisfies 8;

// expressions / comptime / scope / comptime values return lowerable data
struct Config {
    width: uint;
    tags: string[];
}

const config = comptime Config {
    width: 4,
    tags: ["fast", "safe"],
};

config.width satisfies 4;
config.tags[0] satisfies "fast";
