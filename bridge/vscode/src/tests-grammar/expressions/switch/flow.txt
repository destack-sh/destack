// expressions / switch / flow / switch breaks target the innermost switch
function invalidSwitchBreakValueInLoop(value: int32): int32 {
// <- keyword.declaration.type.ds
    loop {
        switch (value) {
            case 0:
                break 1;
            default:
                break;
        }
        break;
    }
    0
}

// expressions / switch / flow / labeled break can target a switch
function labeledBreakSwitch(value: int32): int32 {
    let count: int32 = 0;
    outer: switch (value) {
        case 0:
            count = 1;
            break :outer;
        default:
            count = 2;
            break;
    }
    count
}

// expressions / switch / flow / labeled break can target a loop
function labeledBreakLoopFromSwitch(value: int32): int32 {
    let count: int32 = 0;
    outer: loop {
        switch (value) {
            case 0:
                count = 1;
                break :outer;
            default:
                count = 2;
                break;
        }
        count = count + 1;
        break;
    }
    count
}
