// expressions / switch / cases / switch selects a matching case
function describe(day: int32): string {
    let label: string = "unknown";
    switch (day) {
        case 0:
            label = "sun";
            break;
        default:
            label = "other";
    }
    label
}

// expressions / switch / cases / switch falls through without break
function classify(day: int32): int32 {
    let result: int32 = 0;
    switch (day) {
        case 0:
        case 6:
            result = 1;
            break;
        case 1:
        case 2:
            result = 2;
            break;
        default:
            result = 3;
    }
    result
}

// expressions / switch / cases / switch rejects break values
function invalidBreak(day: int32): int32 {
    switch (day) {

}
