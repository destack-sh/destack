// SYNTAX TEST "source.ds"

// expressions / literals / boolean / true literal
const x: boolean = true;

// expressions / literals / boolean / false literal
const x: boolean = false;

// expressions / literals / boolean / true literals stay exact
const x = true;
x satisfies true;
x satisfies boolean;

// expressions / literals / boolean / false literals stay exact
const x = false;
x satisfies false;
x satisfies boolean;

// expressions / literals / boolean / boolean assigned to string
const x: string = true;

// expressions / literals / boolean / boolean assigned to number
const x: number = false;

// expressions / literals / boolean / boolean does not satisfy string
const x = true;
x satisfies string;

// expressions / literals / boolean / boolean toString resolves
const value = true;
const text = value.toString();
text satisfies string;
