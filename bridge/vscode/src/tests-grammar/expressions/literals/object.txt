// SYNTAX TEST "source.ds"

// expressions / literals / object / object with properties
const x = { a: 1, b: "two" };
x satisfies { a: number; b: string };

// expressions / literals / object / object with shorthand properties
const name = "Ada";
const age = 42;
const person = { name, age };
person satisfies { name: string; age: number };

// expressions / literals / object / empty object
const x = {};
x satisfies {};

// expressions / literals / object / nested object
const x = { a: { b: 1 } };
x satisfies { a: { b: number } };

// expressions / literals / object / object with const assertion
const x = { a: 1, b: "two" } as const;
x satisfies { readonly a: 1; readonly b: "two" };

// expressions / literals / object / object spread adds fields
const base = { a: 1, b: "two" };
const value = { ...base, c: true };
value satisfies { a: number; b: string; c: boolean };

// expressions / literals / object / object spread overrides fields
const base = { a: 1, b: 2 };
const value = { ...base, b: "two" };
value satisfies { a: number; b: string };

// expressions / literals / object / object spread preserves unions
const value: { a: number } | { b: string } = { a: 1 };
const merged = { ...value };
merged satisfies { a: number } | { b: string };

// expressions / literals / object / object spread unwraps aliases
type Base = { a: number };
const base: Base = { a: 1 };
const value = { ...base, b: "two" };
value satisfies { a: number; b: string };

// expressions / literals / object / object spread merges intersections
const base: { a: number } & { b: string } = { a: 1, b: "two" };
const value = { ...base, c: true };
value satisfies { a: number; b: string; c: boolean };

// expressions / literals / object / contextual object literal
const value: { a: number; b: string } = { a: 1, b: "hi" };
value satisfies { a: number; b: string };

// expressions / literals / object / contextual object literal mismatch
const value: { a: number; b: string } = { a: 1, b: 2 };

// expressions / literals / object / contextual object literal via alias
type Point = { x: number; y: number };

const value: Point = { x: 1, y: 2 };
value satisfies { x: number; y: number };

// expressions / literals / object / contextual object literal alias mismatch
type Point = { x: number; y: number };

const value: Point = { x: 1, y: "hi" };

// expressions / literals / object / contextual object spread literal
const value: { a: number } = { ...{ a: 1 } };
value satisfies { a: number };

// expressions / literals / object / contextual object spread literal mismatch
const value: { a: number } = { ...{ a: "hi" } };

// expressions / literals / object / object toString resolves
const value = { a: 1, b: "two" };
const text = value.toString();
text satisfies string;

// expressions / literals / object / object hasOwnProperty resolves
const value = { a: 1 };
const result = value.hasOwnProperty("a");
result satisfies boolean;
