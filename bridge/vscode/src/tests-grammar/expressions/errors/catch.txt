// expressions / errors / catch / catch sees result errors
declare function readConfig(): Result<int, string>;
// <- keyword.declaration.type.ds

const value = try {
    readConfig()?;
    1
} catch (e) {
    e satisfies string;
    2
};
value satisfies int;

// expressions / errors / catch / catch sees unioned errors
declare function readConfig(): Result<int, "missing"> | Result<int, "bad">;

const value = try {
    readConfig()?;
    1
} catch (e) {
    e satisfies "missing" | "bad";
    2
};
value satisfies int;

// expressions / errors / catch / catch sees multiple errors
declare function readConfig(): Result<int, "missing">;

const value = try {
    readConfig()?;
    readVersion()?;
    1
} catch (e) {
    e satisfies "missing" | "bad";
    0
};
value satisfies int;

// expressions / errors / catch / catch can stop custom failures
struct Maybe<T, E> {
    branchValue: TryBranch<T, E>;
}

extension<T, E> of Maybe<T, E> implements Try {
    type Value = T;
    type Failure = E;

    static fromValue(value: T): Maybe<T, E> {
        Maybe { branchValue: TryContinue { kind: "continue", value } }
    }

    branch(): TryBranch<T, E> {
        this.branchValue
    }
}

declare function getMaybe(): Maybe<int, string>;

const value = try {
    getMaybe()?;
    1
} catch (e) {
    e satisfies string;
    0
};
value satisfies int;
