// SYNTAX TEST "source.ds"

// expressions / errors / flow / try expressions require catch or finally
const value = try { 1 };
// <- keyword.declaration.type.ds
value satisfies int;

// expressions / errors / flow / try finally returns the body type
const value = try {
    1
} finally {
    2;
};
value satisfies int;

// expressions / errors / flow / try catch returns union type
const value = try {
    1
} catch (e) {
    e satisfies unknown;
    "fallback"
};
value satisfies int | string;

// expressions / errors / flow / try catch finally returns union type
const value = try {
    1
} catch (e) {
    e satisfies unknown;
    "fallback"
} finally {
    2;
};
value satisfies int | string;
