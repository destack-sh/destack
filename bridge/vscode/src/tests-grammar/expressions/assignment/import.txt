// expressions / assignment / import / imported bindings are immutable
export let counter: number = 0;
// <- keyword.declaration.import.ds

// expressions / assignment / import / imported bindings are immutable
import { counter } from "./counter.ds";

counter = 1;

// expressions / assignment / import / imported aliases are immutable
export let counter: number = 0;

// expressions / assignment / import / imported aliases are immutable
import { counter as localCounter } from "./counter.ds";

localCounter = 1;

// expressions / assignment / import / imported namespace bindings are immutable
export let counter: number = 0;

// expressions / assignment / import / imported namespace bindings are immutable
import * as counter from "./counter.ds";

counter = { counter: 1 };

// expressions / assignment / import / imported namespace members are immutable
export let counter: number = 0;

// expressions / assignment / import / imported namespace members are immutable
import * as namespaceCounter from "./counter.ds";

namespaceCounter.counter = 1;

// expressions / assignment / import / imported bindings reject updates
export let counter: number = 0;

// expressions / assignment / import / imported bindings reject updates
import { counter } from "./counter.ds";

counter++;
