// SYNTAX TEST "source.ds"

// expressions / ranges / subscript / range indexing returns a slice view
declare const values: Slice<int32>;

const middle = values[1..4];
middle satisfies Slice<int32>;

// expressions / ranges / subscript / readonly range indexing preserves access
declare const values: Slice<int32>;

const middle = (&readonly values)[1..4];
middle satisfies &readonly Slice<int32>;

// expressions / ranges / subscript / exclusive range indexing preserves access
declare const values: Slice<int32>;

const middle = (&exclusive values)[1..4];
middle satisfies &exclusive Slice<int32>;

// expressions / ranges / subscript / full range indexing returns the whole view
declare const values: Slice<int32>;

const all = values[..];
all satisfies Slice<int32>;

// expressions / ranges / subscript / one-sided range indexing returns a slice view
declare const values: Slice<int32>;

const tail = values[2..];
const head = values[..4];
const prefix = values[..=4];

tail satisfies Slice<int32>;
head satisfies Slice<int32>;
prefix satisfies Slice<int32>;

// expressions / ranges / subscript / inclusive range indexing uses inclusive bounds
declare const values: Slice<int32>;

const middle = values[1..=3];
middle satisfies Slice<int32>;

// expressions / ranges / subscript / range indexing accepts inferred endpoint types
declare const values: Slice<int32>;
declare const start: usize;
declare const end: usize;

const middle = values[start..end];
middle satisfies Slice<int32>;

// expressions / ranges / subscript / range writes copy into the selected slice
declare let values: Slice<int32>;

values[1..4] = [7, 8, 9];

// expressions / ranges / subscript / range writes accept readonly source slices
declare let target: Slice<int32>;
declare const source: Slice<int32>;

target[1..4] = (&readonly source)[0..3];

// expressions / ranges / subscript / range writes require copyable elements
class File {}

declare function openFile(path: string): ^File;

declare let files: Slice<^File>;

files[1..2] = [openFile("log.txt")];

// expressions / ranges / subscript / readonly range writes are rejected
declare const values: &readonly Slice<int32>;

values[1..4] = [7, 8, 9];
