// SYNTAX TEST "source.ds"

// expressions / ranges / values / half-open ranges
const range = 0..10;
range satisfies Range<int>;

// expressions / ranges / values / inclusive ranges
const range = 0..=10;
range satisfies RangeInclusive<int>;

// expressions / ranges / values / one-sided ranges
const from = 2..;
const to = ..10;
const through = ..=10;

from satisfies RangeFrom<int>;
to satisfies RangeTo<int>;
through satisfies RangeToInclusive<int>;

// expressions / ranges / values / full ranges
const range = ..;
range satisfies RangeFull;

// expressions / ranges / values / endpoints must share one type
const range = 0..10n;

// expressions / ranges / values / inclusive ranges keep their endpoint value
const last: uint8 = 255;
const range = 0..=last;

range satisfies RangeInclusive<uint8>;

// expressions / ranges / values / float endpoints are allowed for range values
const range = 0.0..1.0;

range satisfies Range<float>;
