// SYNTAX TEST "source.ds"

// expressions / annotations / derive / derive applies to structs
@derive(Clone)
// <- meta.decorator.call.ds
struct User {
    id: int64;
}

const user = User { id: 1 };
user.clone() satisfies User;

// expressions / annotations / derive / derive can request multiple providers
@derive(Clone, Debug)
struct User {
    id: int64;
}

const user = User { id: 1 };
user.clone() satisfies User;
user.debug() satisfies string;

// expressions / annotations / derive / module derive providers run automatically
module {
    derive: [Clone],
};

struct User {
    id: int64;
}

const user = User { id: 1 };
user.clone() satisfies User;
