// expressions / match / flow / match does not allow break
function invalidMatchBreak(value: int32): int32 {
    const result = match (value) {
        0 => break
        _ => 1
    };
    result
}

// expressions / match / flow / match does not allow continue
function invalidMatchContinue(value: int32): int32 {
    const result = match (value) {
        0 => continue
        _ => 1
    };
    result
}

// expressions / match / flow / nested loops allow break
function nestedLoopBreak(value: int32): int32 {
    let result: int32 = 0;
    match (value) {
        0 => {
            loop {
                break;
            }
            result = 1;
        }
        _ => {
            result = 2;
        }
    }
    result
}

// expressions / match / flow / nested loops allow continue
function nestedLoopContinue(value: int32): int32 {
    let result: int32 = 0;
    match (value) {
        0 => {
            let current: int32 = 0;
            while (current < 2) {
                current = current + 1;
                continue;
            }
            result = current;
        }
        _ => {
            result = 3;
        }
    }
    result
}
