// expressions / if / branches / if (let ...) joins branches
declare const value: int32;
// <- keyword.declaration.type.ds

const result = if (let 1 = value) {
    "one"
} else {
    2
};

result satisfies string | int32;

// expressions / if / branches / if (let ...) chains with else if
declare const value: 1 | 2 | 3;

const result = if (let 1 = value) {
    "one"
} else if (let 2 = value) {
    "two"
} else {
    "three"
};

result satisfies string;

// expressions / if / branches / if (let ...) without else yields void
declare const value: int32;

const result: int32 = if (let 1 = value) {
    1
};

// expressions / if / branches / if (let ...) bindings stay in the then branch
declare const pair: (int32, int32);

const result = if (let (left, right) = pair) {
    left + right
} else {
    left
};

// expressions / if / branches / if (let ...) bindings do not escape the if expression
declare const value: int32;

if (let x = value) {
    x
}

x;

// expressions / if / branches / if (let ...) narrows both branches
declare const value: 1 | 2;

if (let 1 = value) {
    value satisfies 1;
} else {
    value satisfies 2;
}

// expressions / if / branches / if (let ...) annotations accept matching values
declare const value: int32;

if (let x: int32 = value) {
    x satisfies int32;
}

// expressions / if / branches / if (let ...) annotations reject mismatches
declare const value: string | int32;

if (let x: int32 = value) {
    x
