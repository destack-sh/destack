// expressions / macros / serde / serde derives satisfy json protocols
import {
    JSON,
    Json,
    JsonDeserialize,
    JsonSerialize,
    decode,
    encode,
} from "destack:json";

@derive(Serialize, Deserialize)
struct ServerConfig {
    host: string;
    port: uint16;
    secure: boolean;
}

const config = decode<ServerConfig>("{\"host\":\"127.0.0.1\",\"port\":8080,\"secure\":true}")?;

config satisfies ServerConfig;
config satisfies JsonSerialize;
ServerConfig satisfies JsonDeserialize;

const text = encode(config)?;
text satisfies string;

const value = JSON.parse(text)?;
value satisfies Json;
JSON.stringify(value)? satisfies string;

// expressions / macros / serde / serde derives compose through fields
import { decode, encode } from "destack:json";

@derive(Serialize, Deserialize)
struct DatabaseConfig {
    url: string;
    poolSize: uint16;
}

@derive(Serialize, Deserialize)
struct AppConfig {
    database: DatabaseConfig;
    tags: string[];
}

const config = decode<AppConfig>(
    "{\"database\":{\"url\":\"postgres://local\",\"poolSize\":8},\"tags\":[\"dev\"]}",
)?;

config.database.poolSize satisfies uint16;
encode(config)? satisfies string;
