// SYNTAX TEST "source.ds"

// expressions / loops / while / while yields void
let value: void = while (true) {
    break;
};

// expressions / loops / while / while is not assignable to number
let value: number = while (true) {
    break;
};

// expressions / loops / while / while allows continue in the loop body
let i = 0;
while (i < 3) {
    i = i + 1;
    if (i == 2) {
        continue;
    }
}

// expressions / loops / while / while allows break in the loop body
let i = 0;
while (i < 3) {
    i = i + 1;
    if (i == 2) {
        break;
    }
}

// expressions / loops / while / while condition supports narrowing in the body
let value: string | int32 = "ok";
while (typeof value == "string") {
    value satisfies string;
    break;
}
