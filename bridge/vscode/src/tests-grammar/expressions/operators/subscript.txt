// expressions / operators / subscript / index access dispatches to Index
struct Bag {
    value: int;
}

extension of Bag implements Index<int> {
    type Output = int;

    index(key: int): this.Output {
        return key;
    }
}

declare function getBag(): Bag;

const bag = getBag();
const value = bag[1];
value satisfies int;
