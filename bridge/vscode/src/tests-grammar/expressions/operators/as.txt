// SYNTAX TEST "source.ds"

// expressions / operators / as / as preserves the asserted type
const value = "ok" as string;
value satisfies string;

// expressions / operators / as / as can narrow unions
const value = ("ok" as string | number) as string;
value satisfies string;

// expressions / operators / as / as can widen to supertypes
const value: 42 = 42;
const widened = value as number;
widened satisfies number;

// expressions / operators / as / as can narrow from unknown
declare const value: unknown;

const narrowed = value as number;
narrowed satisfies number;

// expressions / operators / as / as can widen object shapes
const value = { x: 1, y: 2 };
const narrowed = value as { x: number };
narrowed satisfies { x: number };

// expressions / operators / as / as rejects unrelated types
const value: string = "hello";
const numberValue = value as number;

// expressions / operators / as / as rejects boolean to number
const value: boolean = true;
const numberValue = value as number;
