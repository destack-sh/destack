// expressions / static if / extension / when false, @if leaves other extension members alone
struct Box {
    value: number;
}

extension of Box {
    @if(false)
    hidden(): number {
        this.value
    }

    get(): number {
        this.value
    }

    increment(): number {
        this.value + 1
    }
}

const box = makeBox();

box.increment() satisfies number;
