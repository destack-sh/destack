// expressions / static if / interface / when false, @if removes interface members
interface Box {
// <- keyword.declaration.type.ds
    @if(false)
    missing: MissingType;
    @if(false)
    missingMethod(): MissingType;

    value: number;
}

declare const box: Box;
box.value satisfies number;

// expressions / static if / interface / when true, @if includes interface members
interface Box {
    @if(true)
    value: number;
    @if(true)
    increment(): number;
}

declare const box: Box;
box.value satisfies number;
box.increment() satisfies number;

// expressions / static if / interface / when false, @if hides interface members
interface Box {
    @if(false)
    missing: number;

    value: number;
}

declare const box: Box;
box.missing satisfies number;

// expressions / static if / interface / when false, @if removes implementation requirements
interface Box {
    @if(false)
    missing(): number;

    value(): number;
}

class Concrete implements Box {
    value(): number {
        1
    }
}

// expressions / static if / interface / when true, @if keeps implementation requirements
interface Box {
    @if(true)
    value(): number;
}

class Concrete implements Box {}
