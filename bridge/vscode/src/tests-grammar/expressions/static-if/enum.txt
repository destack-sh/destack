// expressions / static if / enum / when false, @if removes enum fields
enum Status {
    @if(false)
    Missing = missingSymbol,
    Visible = 1,
}

// expressions / static if / enum / when true, @if includes enum fields
enum Status {
    @if(true)
    Active = 1,
    Inactive = 2,
}

const active = Status.Active;
active satisfies Status;

    @if(false)
    Missing = 0,

// expressions / static if / enum / when false, @if leaves other enum fields alone

    @if(false)

    Ready = 1,
    Done = 2,
