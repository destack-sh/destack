// SYNTAX TEST "source.ds"

// memory / owned / owned expression yields owned value
struct Point {
    x: int32;
}

let point = ^Point { x: 1 };

point satisfies ^Point;

// memory / owned / plain values do not satisfy owned destinations
struct Point {
    x: int32;
}

let point: ^Point = Point { x: 1 };

// memory / owned / owned parameters consume owned values
struct Point {
    x: int32;
}

function consume(point: ^Point): void {
    point.x;
}

let point = ^Point { x: 1 };
consume(point);
point.x;

// memory / owned / owned conversion moves plain values
struct Point {
    x: int32;
}

function consume(point: ^Point): void {
    point.x;
}

let point = Point { x: 1 };
consume(^point);

// memory / owned / owned conversion rejects owned values
struct Point {
    x: int32;
}

let point = ^Point { x: 1 };
let again = ^point;

// memory / owned / structs can store owned fields
struct Data {
    value: int32;
}

struct Container {
    data: ^Data;
}

const container = Container { data: ^Data { value: 1 } };

container.data satisfies ^Data;

// memory / owned / readonly owned values forbid mutation
struct Point {
    x: int32;
}

let point: ^readonly Point = ^readonly Point { x: 1 };
point.x = 2;

// memory / owned / readonly owned values are deep
struct Profile {
    name: string;
}

struct User {
    profile: Profile;
}

let user: ^readonly User = ^readonly User {
    profile: Profile { name: "Ada" },
};

user.profile.name = "Grace";
