// SYNTAX TEST "source.ds"

// memory / new / new follows managed destinations
class Widget {
// <- keyword.declaration.type.ds
    value: int32 = 0;
}

let box: Widget = new Widget();

box satisfies Widget;

// memory / new / new follows owned destinations
class Widget {
    value: int32 = 0;
}

let box: ^Widget = new Widget();

box satisfies ^Widget;

// memory / new / new follows owned shared destinations
class Widget {
    value: int32 = 0;
}

let box: shared ^Widget = new Widget();

box satisfies shared ^Widget;
box satisfies ^shared Widget;

// memory / new / new follows shared destinations
class Widget {
    value: int32 = 0;
}

let box: shared Widget = new Widget();

box satisfies shared Widget;

// memory / new / ambient fields follow the destination
class Header {
    id: int32 = 0;
}

class Payload {
    value: int32 = 0;
}

class Request<T> {
    header: Header = new Header();
    body: T;

    constructor(body: T) {
        this.body = body;
    }
}

let request: shared Request<Payload> = new Request(new Payload());

request.header satisfies shared Header;
request.body satisfies shared Payload;

// memory / new / address expression can borrow
class Widget {
    value: int32 = 0;
}

let box = new Widget();
let borrow: &Widget = &box;

borrow satisfies &Widget;

// memory / new / address expression can create raw pointers
class Widget {
    value: int32 = 0;
}

let box = new Widget();
let pointer: *Widget = &box;

pointer satisfies *Widget;
