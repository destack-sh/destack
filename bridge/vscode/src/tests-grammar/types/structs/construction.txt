// types / structs / construction / tagged literals construct structs
struct Counter {
// <- keyword.declaration.type.ds
    value: int32;

    increment(): Counter {
        Counter { value: this.value + 1 }
    }
}

const counter = Counter { value: 1 };
const next = counter.increment();
next satisfies Counter;

// types / structs / construction / tagged literals expose methods
struct Counter {
    value: int32;

    increment(): Counter {
        Counter { value: this.value + 1 }
    }
}

const value = (Counter { value: 1 }).increment().value;
value satisfies int32;
