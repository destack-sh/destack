// types / type operators / keys / keyof builds literal key unions
interface Person {
    name: string;
    age: number;
}

type Keys = keyof Person;

const name: Keys = "name";
const age: Keys = "age";

// types / type operators / keys / keyof rejects unknown keys
interface Person {
    name: string;
    age: number;
}

type Keys = keyof Person;

const bad: Keys = "title";

// types / type operators / keys / keyof unions keep shared keys
type Left = { shared: string; left: int32 };
type Right = { shared: string; right: int32 };

type Keys = keyof (Left | Right);

const ok: Keys = "shared";

// types / type operators / keys / keyof unions reject missing keys
type Left = { shared: string; left: int32 };
type Right = { shared: string; right: int32 };

type Keys = keyof (Left | Right);

const bad: Keys = "left";

// types / type operators / keys / keyof intersections include all keys
type Left = { shared: string; left: int32 };
type Right = { shared: string; right: int32 };

type Keys = keyof (Left & Right);

const shared: Keys = "shared";
const left: Keys = "left";
const right: Keys = "right";

// types / type operators / keys / keyof uses instantiated keys
type Keys<T: { a: number }> = keyof T;
type Actual = Keys<{ a: number; b: string }>;

const key: Actual = "b";
key satisfies "a" | "b";
