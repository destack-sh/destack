// types / generics / structs / structs accept explicit type arguments
struct Box<T> {
// <- keyword.declaration.type.ds
    value: T;
}

declare function makeBox(): Box<number>;

let value: Box<number> = makeBox();
value satisfies Box<number>;

// types / generics / structs / structs reject type argument mismatches

    value: T;
}

// types / generics / structs / structs accept static value arguments

// types / generics / structs / structs reject static value argument mismatches
