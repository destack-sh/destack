// types / generics / rest / partial application preserves tail tuple order
declare function partial<T: readonly unknown[], U: readonly unknown[], R>(
    fn: (...args: [...T, ...U]) => R,
    ...head: T
): (...tail: U) => R;

declare function join(a: "a", b: 1, c: true): "ok";

const tail = partial(join, "a");
const value = tail(1, true);
value satisfies "ok";

// types / generics / rest / partial application rejects reordered tail arguments
declare function partial<T: readonly unknown[], U: readonly unknown[], R>(

    ...head: T
): (...tail: U) => R;

declare function join(a: "a", b: 1, c: true): "ok";

const tail = partial(join, "a");
tail(true, 1);

// types / generics / rest / const rest inference preserves literal tuple elements
declare function collect<const T: readonly unknown[]>(...values: T): T;

const value = collect("x", 1, true);
value[0] satisfies "x";
value[1] satisfies 1;
value[2] satisfies true;

// types / generics / rest / mutable rest inference widens scalar tuple elements
declare function collect<T: readonly unknown[]>(...values: T): T;

let first = "x";
const value = collect(first, 1, true);
value[0] satisfies "x";
