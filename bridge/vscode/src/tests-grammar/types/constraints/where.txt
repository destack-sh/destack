    read(): void;
}

}

    read() {},
};
clone<Readable>(value);

    read(): void;
}

interface Mergeable {
    merge(): void;
}

    value;
    other;

}

const other: Mergeable = {
    merge() {},
};

merge<Readable, Mergeable>(value, other);
