// SYNTAX TEST "source.ds"

// types / primitives / unknown / values assign to unknown
const value: unknown = 42;
value satisfies unknown;

// types / primitives / unknown / unknown rejects arbitrary member access
declare const value: unknown;
const result = value.missing;

// types / primitives / unknown / any is rejected
const value: any = 42;

// types / primitives / unknown / any does not enable dynamic access
declare const value: any;
const result = value.missing.member;
