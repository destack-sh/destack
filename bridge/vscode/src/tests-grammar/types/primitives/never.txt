// types / primitives / never / never to number

// <- keyword.declaration.type.ds
    panic("error");
}

// types / primitives / never / never to string
function fail(): never {
    panic("error");
}
const x: string = fail();

// types / primitives / never / number to never is rejected
const x: never = 1;

// types / primitives / never / string to never is rejected
const x: never = "no";

// types / primitives / never / union with never simplifies to the other member

const x: Value = "ok";

// types / primitives / never / intersection with never rejects all concrete values
