// SYNTAX TEST "source.ds"

// types / extensions / calls / extension methods return primitives
struct Point {
// <- keyword.declaration.type.ds
    x: number;
    y: number;
}

extension of Point {
    length(): number {
        return 0;
    }
}

declare function getPoint(): Point;

const point = getPoint();
point.length() satisfies number;

// types / extensions / calls / extension methods return void
struct Logger {
    prefix: string;
}

extension of Logger {
    log(message: string): void {}
}

declare function getLogger(): Logger;

const logger = getLogger();
logger.log("hello");

// types / extensions / calls / extension methods return the receiver type
struct Vector2 {
    x: number;
    y: number;
}

extension of Vector2 {
    normalized(): Vector2 {
        return Vector2 { x: 0, y: 0 };
    }
}

declare function getVector(): Vector2;

const vector = getVector();
vector.normalized() satisfies Vector2;

// types / extensions / calls / extension methods return other types
struct Point {
    x: number;
    y: number;
}

extension of Point {

        return "";
    }
}

declare function getPoint(): Point;

const point = getPoint();

// types / extensions / calls / extension methods accept parameters
struct Point {
    x: number;
    y: number;
}

extension of Point {
    scale(factor: number): Point {
        return Point { x: 0, y: 0 };
    }
}

declare function getPoint(): Point;

const point = getPoint();
point.scale(2) satisfies Point;

// types / extensions / calls / extension methods accept multiple parameters
struct Point {
    x: number;
    y: number;
}

extension of Point {
    translate(dx: number, dy: number): Point {
        return Point { x: 0, y: 0 };
    }
}

declare function getPoint(): Point;

const point = getPoint();
point.translate(10, 20) satisfies Point;

// types / extensions / calls / extension methods accept optional parameters
struct Logger {
    prefix: string;
}

extension of Logger {
    log(message: string, level?: number): void {}
}

declare function getLogger(): Logger;

const logger = getLogger();
logger.log("info");
logger.log("warn", 2);

// types / extensions / calls / extension methods can constrain this
struct Counter {
    value: number;
}

extension of Counter {
    increment(this: &Counter): void {
        this.value = this.value + 1;
    }
}

let counter = Counter { value: 0 };
counter.increment();

// types / extensions / calls / extension methods chain through return types
struct StringBuilder {
    value: string;
}

extension of StringBuilder {
    append(text: string): StringBuilder {
        return StringBuilder { value: "" };
    }
}

declare function getStringBuilder(): StringBuilder;

const builder = getStringBuilder();
builder.append("a").append("b").append("c") satisfies StringBuilder;
