// SYNTAX TEST "source.ds"

// types / convert / from / From converts into the implementing type
newtype UserId = string;

extension of UserId implements From<string> {
    static from(value: string): UserId {
        return UserId(value);
    }
}

const id = UserId.from("u_123");
id satisfies UserId;

// types / convert / from / Into is provided by From
newtype UserId = string;

extension of UserId implements From<string> {
    static from(value: string): UserId {
        return UserId(value);
    }
}

const source = "u_123";
const id: UserId = source.into();
id satisfies UserId;

// types / convert / from / Into does not chain through intermediate conversions
newtype UserId = string;
newtype AccountId = UserId;

extension of UserId implements From<string> {
    static from(value: string): UserId {
        return UserId(value);
    }
}

extension of AccountId implements From<UserId> {
    static from(value: UserId): AccountId {
        return AccountId(value);
    }
}

const source = "u_123";
const id: AccountId = source.into();
