// SYNTAX TEST "source.ds"

// types / convert / reference / As defaults to readonly access
class Buffer {
// <- keyword.declaration.type.ds
    bytes: uint8[];
}

extension of Buffer implements As<[uint8]> {
    as(): &readonly [uint8] {
        return &readonly this.bytes;
    }
}

declare function read(bytes: &readonly [uint8]): void;

const buffer = new Buffer();
read(buffer.as());

// types / convert / reference / As selects access explicitly
class Buffer {
    bytes: uint8[];
}

extension of Buffer implements As<[uint8], "exclusive"> {
    as(): &exclusive [uint8] {
        return &exclusive this.bytes;
    }
}

declare function write(bytes: &exclusive [uint8]): void;

const buffer = new Buffer();
write(buffer.as());
