// SYNTAX TEST "source.ds"

// types / intervals / scalars / char intervals use Unicode scalar order
type LowerAscii = 'a'..='z';

let value: LowerAscii = 'm';
value satisfies LowerAscii;
value satisfies char;

// types / intervals / scalars / char intervals reject values outside their bounds
type LowerAscii = 'a'..='z';

let value: LowerAscii = 'A';

// types / intervals / scalars / bigint intervals use integer order
type BigId = 1n..=9007199254740993n;

let value: BigId = 9007199254740993n;
value satisfies BigId;
value satisfies bigint;

// types / intervals / scalars / float intervals are rejected because floats are partially ordered
type UnitFloat = 0.0..=1.0;

// types / intervals / scalars / string intervals are rejected because string ordering is not primitive
type Lower = "a"..="z";
