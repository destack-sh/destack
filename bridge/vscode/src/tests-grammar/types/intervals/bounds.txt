// types / intervals / bounds / half-open intervals include the start and exclude the end
type Window = 0..10;
// <- keyword.declaration.type.ds

let value: Window = 5;
value satisfies Window;
value satisfies int;

// types / intervals / bounds / inclusive intervals include both bounds
type Window = 0..=10;

let value: Window = 10;
value satisfies Window;
value satisfies int;

// types / intervals / bounds / one-sided intervals constrain one side

// types / intervals / bounds / interval types reject values outside their bounds

// types / intervals / bounds / interval types compose with unions
