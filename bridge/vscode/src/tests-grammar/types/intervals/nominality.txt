// SYNTAX TEST "source.ds"

// types / intervals / nominality / newtypes can use interval backing types
newtype Port = 1..=65535;

let port = Port(443);
port satisfies Port;

// types / intervals / nominality / validated newtypes preserve runtime invariants through mutation
newtype Port = int;

extension of Port {
    static try(value: int): Result<Port, Error> {
        if (value < 1 || value > 65535) {
            return Result.err(Error.message("port out of range"));
        }

        return Result.ok(Port(value));
    }
}
