// types / objects / assignability / object literal satisfies interface
interface Person {
    name: string;
}

const person = { name: "Ada" };
person satisfies Person;

// types / objects / assignability / missing required field is not assignable

    name: string;
    age: number;
}

// types / objects / assignability / nominal interface requires explicit implements
newtype interface Add<T> {
    type Output;

    add(other: T): this.Output;
}

struct Vec2 {
    x: float32;
    y: float32;

    add(other: Vec2): Vec2 {
        Vec2 { x: this.x + other.x, y: this.y + other.y }
    }
}

const value: Add<Vec2> = Vec2 { x: 1, y: 2 };

// types / objects / assignability / nominal interface accepts explicit implements
newtype interface Add<T> {
    type Output;

    add(other: T): this.Output;
}

struct Vec2 {
    x: float32;
    y: float32;
}

extension of Vec2 implements Add<Vec2> {
    type Output = Vec2;

    add(other: Vec2): this.Output {
        Vec2 { x: this.x + other.x, y: this.y + other.y }
    }
}

const value: Add<Vec2> = Vec2 { x: 1, y: 2 };
value.add(Vec2 { x: 2, y: 3 }) satisfies Vec2;
