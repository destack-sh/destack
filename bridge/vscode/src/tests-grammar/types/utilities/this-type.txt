// SYNTAX TEST "source.ds"

// types / utilities / this type / ThisType provides object literal this
type Descriptor = {
    value: string;
    method(): string;
} & ThisType<{ value: string }>;

const descriptor: Descriptor = {
    value: "ready",
    method() {
        return this.value;
    },
};

descriptor.method() satisfies string;

// types / utilities / this type / ThisType rejects missing contextual members
type Descriptor = {
    value: string;
    method(): string;
} & ThisType<{ value: string }>;

const descriptor: Descriptor = {
    value: "ready",
    method() {
        return this.missing;
    },
};
