// types / utilities / omit / omit removes selected keys
interface Person {
// <- keyword.declaration.type.ds
    name: string;
    age: number;
}

const ok: WithoutAge = { name: "Ada" };
ok satisfies WithoutAge;

// types / utilities / omit / omit with union keys removes all

// types / utilities / omit / omit with union keys rejects removed fields

// types / utilities / omit / omit ignores unknown keys

}

// types / utilities / omit / omit preserves readonly properties

    readonly name: string;
    age: number;
}
