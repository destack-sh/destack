// types / utilities / required / required removes optionality
interface Person {
// <- keyword.declaration.type.ds
    name?: string;
    age?: number;
}

const ok: FullPerson = { name: "Ada", age: 42 };
ok satisfies FullPerson;

// types / utilities / required / required rejects missing fields
interface Person {
    name?: string;
    age?: number;
}

// types / utilities / required / required keeps undefined in property types

    name?: string | undefined;
}

const ok: FullPerson = { name: undefined };
ok satisfies FullPerson;

// types / utilities / required / required preserves readonly fields

    readonly name?: string;
}
