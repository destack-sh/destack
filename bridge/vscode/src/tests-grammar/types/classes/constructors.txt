// types / classes / constructors / required fields need initialization
class Counter {
    value: number;
}

// types / classes / constructors / constructors initialize required fields
class Counter {
    value: number;

    constructor(value: number) {
        this.value = value;
    }
}

// types / classes / constructors / optional fields do not need initialization
class Counter {
    value?: number;
}

// types / classes / constructors / all constructor paths must initialize fields
class Counter {
    value: number;

    constructor(flag: boolean) {
        if (flag) {
            this.value = 1;
        }
    }
}

// types / classes / constructors / constructor overloads share one implementation
class Box {
    value: string | number;

    constructor(value: string);
    constructor(value: number);
    constructor(value: string | number) {
        this.value = value;
    }
}

const fromString = new Box("x");
const fromNumber = new Box(42);

fromString.value satisfies string | number;
fromNumber.value satisfies string | number;

// types / classes / constructors / constructor overloads reject multiple implementations
class Box {
    value: string | number;

    constructor(value: string) {
        this.value = value;
    }

    constructor(value: number) {
        this.value = value;
    }
}

// types / classes / constructors / constructor overloads require a compatible implementation
class Box {

    constructor(value: number);
    constructor(value: boolean) {
        this.value = value ? 1 : 0;
    }
}
