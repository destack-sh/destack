    add(other: T): this.Output;

struct Vec2 {
    x: int32;
    y: int32;

    add(other: Vec2): Vec2 {
        Vec2 { x: this.x + other.x, y: this.y + other.y }
    }
}

// types / newtypes / interfaces / explicit implements satisfies nominal interfaces
newtype interface Add<T> {
    type Output;

    add(other: T): this.Output;
}

struct Vec2 {
    x: int32;
    y: int32;
}

extension of Vec2 implements Add<Vec2> {
    type Output = Vec2;

    add(other: Vec2): this.Output {
        Vec2 { x: this.x + other.x, y: this.y + other.y }
    }
}
