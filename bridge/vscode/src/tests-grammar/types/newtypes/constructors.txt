// SYNTAX TEST "source.ds"

// types / newtypes / constructors / scalar newtypes use one argument
newtype UserId = int64;
// <- keyword.declaration.type.ds

const id = UserId(42);
id satisfies UserId;

// types / newtypes / constructors / tuple newtypes use positional arguments
newtype Point = (float32, float32);

const point = Point(1.0, 2.0);
point satisfies Point;

// types / newtypes / constructors / object newtypes use one object argument
newtype Config = { debug: boolean };

const config = Config({ debug: true });
config satisfies Config;

// types / newtypes / constructors / object literals do not construct newtypes
newtype Config = { debug: boolean };

const config: Config = { debug: true };
