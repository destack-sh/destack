// SYNTAX TEST "source.ds"

// types / enums / backing / integer enums do not coerce to integers
enum Status {
    Active = 1,
    Inactive = 2,
}

const status = Status.Active;
const raw: int32 = status;

// types / enums / backing / integer enums cast to integers explicitly
enum Status {
    Active = 1,
    Inactive = 2,
}

const status = Status.Active;
const raw = status as int32;
raw satisfies int32;

// types / enums / backing / integers cast to enums explicitly
enum Status {
    Active = 1,
    Inactive = 2,
}

const status = 1 as Status;
status satisfies Status;

// types / enums / backing / integers do not coerce to enums
enum Status {
    Active = 1,
    Inactive = 2,
}

const status: Status = 1;

// types / enums / backing / string enums do not coerce to strings
enum Flavor {
    Sweet = "sweet",
    Sour = "sour",
}

const flavor = Flavor.Sweet;
const raw: string = flavor;

// types / enums / backing / string enums cast to strings explicitly
enum Flavor {
    Sweet = "sweet",
    Sour = "sour",
}

const flavor = Flavor.Sweet;
const raw = flavor as string;
raw satisfies string;

// types / enums / backing / strings cast to enums explicitly
enum Flavor {
    Sweet = "sweet",
    Sour = "sour",
}

const flavor = "sweet" as Flavor;
flavor satisfies Flavor;

// types / enums / backing / enum members share one backing type
enum Mixed {
    First = 1,
    Second = "two",
}
