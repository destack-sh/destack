// SYNTAX TEST "source.ds"

// runtime / modules / data / JSON object imports resolve
import data from "./data.json.ds";

data;

// runtime / modules / data / JSON array imports resolve
import numbers from "./numbers.json.ds";

numbers;

// runtime / modules / data / override loader with type attribute
import text from "./data.json.ds" with { type: "text" };

// imported as text (string), not as JSON object
text;

// runtime / modules / data / same file with different loaders produces different modules
import json from "./data.json.ds";
import text from "./data.json.ds" with { type: "text" };

json;
text;

// runtime / modules / data / re-export supports type attribute loader overrides
export { default as text } from "./data.json.ds" with { type: "text" };

// runtime / modules / data / re-export supports type attribute loader overrides
import { text } from "./forward.ds";

text satisfies string;

// runtime / modules / data / unknown type attribute reports an error
import data from "./data.json.ds" with { type: "jsonc" };

data;

// runtime / modules / data / property access on imported JSON is typed
import data from "./data.json.ds";

data.name satisfies "Alice";
data.age satisfies 30;

// runtime / modules / data / nested object property access is typed
import config from "./config.json.ds";

config.server.host satisfies "localhost";
config.server.port satisfies 8080;
config.debug satisfies true;

// runtime / modules / data / array imports keep tuple shape
import numbers from "./numbers.json.ds";

numbers satisfies readonly (1, 2, 3);
numbers[0] satisfies 1;
numbers[2] satisfies 3;

// runtime / modules / data / imported JSON can be widened
interface User {
    name: string;
    age: number;
}

import data from "./data.json.ds";

const user: User = data;
user.name satisfies string;

// runtime / modules / data / nonexistent property access is error
import data from "./data.json.ds";

data.nonexistent;

// runtime / modules / data / base64 import is typed as string
import encoded from "./data.txt.ds" with { type: "base64" };

encoded satisfies string;

// runtime / modules / data / namespace import from JSON is typed
import * as data from "./data.json.ds";

data.name satisfies "Alice";
data.age satisfies 30;
