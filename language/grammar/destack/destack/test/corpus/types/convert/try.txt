struct ParseIdError {
    message: string;
}

newtype UserId = string;

extension of UserId implements TryFrom<string, ParseIdError> {
    static tryFrom(value: string): Result<UserId, ParseIdError> {
        return Result.ok(UserId(value));
    }
}

const id = UserId.tryFrom("u_123")?;
id satisfies UserId;
