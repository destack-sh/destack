struct Pair<A, B> {
    left: A;
    right: B;
}

extension<Left, Right> of Pair<Right, Left> {
    swap(): Pair<Left, Right> {
        return Pair<Left, Right> {
            left: this.right,
            right: this.left,
        };
    }
}

declare function makePair(): Pair<number, string>;

const pair = makePair();
pair.swap() satisfies Pair<string, number>;
