interface Readable {
    read(): void;
}

interface Mergeable {
    merge(): void;
}

function merge<T, U>(value: T, other: U): T where (T: Readable, U: Mergeable) {
    value;
    other;
    return value;
}

const value: Readable = {
    read() {},
};
const other: Mergeable = {
    merge() {},
};

merge<Readable, Mergeable>(value, other);
