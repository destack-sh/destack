        return Vector2 { x: 0, y: 0 };
    }
}

    scale(factor: number): Vector2 {
        return Vector2 { x: 0, y: 0 };
    }
}

declare function getVector(): Vector2;

const vector = getVector();
vector.add(vector) satisfies Vector2;
vector.scale(2) satisfies Vector2;
