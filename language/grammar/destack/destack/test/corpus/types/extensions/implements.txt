interface Describable {
    describe(): string;
}

struct Point {
    x: number;
    y: number;
}

extension of Point implements Describable {
    describe(): string {
        return "";
    }
}

declare function getPoint(): Point;

const point = getPoint();
point satisfies Describable;
point.describe() satisfies string;

    print(): void;
}

interface Serializable {
    serialize(): string;
}

struct Document {
    content: string;
}

extension of Document implements Printable, Serializable {
    print(): void {}
    serialize(): string {
        return "";
    }
}

declare function getDocument(): Document;

const document = getDocument();
document satisfies Printable;
document satisfies Serializable;
document.print();
document.serialize() satisfies string;
