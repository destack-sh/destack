let value = token.intoInner();

===============================================
memory / pin / inline self borrows are rejected
===============================================

class User {
    name: string = "";
}

struct NameView<L: Lifetime> {
    user: ^User;
    name: Borrowed<string, L>;
}

function makeView<L: Lifetime>(): NameView<L> {
    let user: ^User = new User();
    return NameView {
        user,
        name: &user.name,
    };
}
