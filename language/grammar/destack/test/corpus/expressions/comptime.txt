function parse(value: int): Result<int, string> {
    if (value > 0) {
        Result.ok(value)
    } else {
        Result.err("bad")
    }
}

const value = comptime {
    try {
        parse(0)?
    } catch (error) {
        error satisfies string;
        1
    }
};

value satisfies int;
